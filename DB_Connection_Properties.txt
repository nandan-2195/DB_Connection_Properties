.yml file for MySQL DB
-----------------------------
spring:
  application:
    name: Spring-Boot_Crud-Operation
  datasource:
    driver-class-name: com.mysql.cj.jdbc.Driver
    password: nandan
    url: jdbc:mysql://localhost:3306/sbms
    username: root
  jpa:
    hibernate:
      ddl-auto: update
    show-sql: true
-------------------------------------------------------------------------------
application.xml file for MySQL DB
-----------------------------
#Change Server
server.port=9091

#Location for presentation file
spring.mvc.view.prefix=/views/
spring.mvc.view.suffix=.jsp

#Define data source properties
spring.datasource.username=root
spring.datasource.password=nandan
spring.datasource.url=jdbc:mysql://localhost:3306/sbms
spring.datasource.driver-class-name=com.mysql.cj.jdbc.Driver

spring.jpa.hibernate.ddl-auto=update
spring.jpa.show-sql=true

#To show our sql query in sql format
sprint.jpa.properties.hibernate.format_sql=true
